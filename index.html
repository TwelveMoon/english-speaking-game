    <script type="text/babel">

        const { useState, useEffect, useRef, useCallback } = React;


        // --- 題庫區 (包含數字與近似音) ---

        const WORDS = [

            { en: 'six', zh: '六', symbol: '6️⃣', aliases: ['6', 'sex'] },

            { en: 'seven', zh: '七', symbol: '7️⃣', aliases: ['7'] },

            { en: 'eight', zh: '八', symbol: '8️⃣', aliases: ['8', 'ate', 'a'] },

            { en: 'nine', zh: '九', symbol: '9️⃣', aliases: ['9', 'line', 'night'] },

            { en: 'ten', zh: '十', symbol: '🔟', aliases: ['10', 'tin', 'pen'] }

        ];


        // --- Icons ---

        const IconStar = ({ size = 24, className = "", fill = "none" }) => (

            <svg xmlns="http://www.w3.org/2000/svg" width={size} height={size} viewBox="0 0 24 24" fill={fill} stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className={className}><polygon points="12 2 15.09 8.26 22 9.27 17 14.14 18.18 21.02 12 17.77 5.82 21.02 7 14.14 2 9.27 8.91 8.26 12 2"></polygon></svg>

        );

        const IconTrophy = ({ size = 24, className = "" }) => (

            <svg xmlns="http://www.w3.org/2000/svg" width={size} height={size} viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className={className}><path d="M6 9H4.5a2.5 2.5 0 0 1 0-5H6"></path><path d="M18 9h1.5a2.5 2.5 0 0 0 0-5H18"></path><path d="M4 22h16"></path><path d="M10 14.66V17c0 .55-.47.98-.97 1.21C7.85 18.75 7 20.24 7 22"></path><path d="M14 14.66V17c0 .55.47.98.97 1.21C16.15 18.75 17 20.24 17 22"></path><path d="M18 2H6v7c0 6 6 10 6 10s6-4 6-10V2z"></path></svg>

        );

        const IconRefreshCcw = ({ size = 24, className = "" }) => (

            <svg xmlns="http://www.w3.org/2000/svg" width={size} height={size} viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className={className}><polyline points="1 4 1 10 7 10"></polyline><polyline points="23 20 23 14 17 14"></polyline><path d="M20.49 9A9 9 0 0 0 5.64 5.64L1 10m22 4l-4.64 4.36A9 9 0 0 1 3.51 15"></path></svg>

        );

        const IconPlay = ({ size = 24, className = "", fill="none" }) => (

            <svg xmlns="http://www.w3.org/2000/svg" width={size} height={size} viewBox="0 0 24 24" fill={fill} stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className={className}><polygon points="5 3 19 12 5 21 5 3"></polygon></svg>

        );

        const IconMic = ({ size = 24, className = "" }) => (

            <svg xmlns="http://www.w3.org/2000/svg" width={size} height={size} viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className={className}><path d="M12 2a3 3 0 0 0-3 3v7a3 3 0 0 0 6 0V5a3 3 0 0 0-3-3Z"></path><path d="M19 10v2a7 7 0 0 1-14 0v-2"></path><line x1="12" y1="19" x2="12" y2="22"></line></svg>

        );

        const IconMicOff = ({ size = 24, className = "" }) => (

            <svg xmlns="http://www.w3.org/2000/svg" width={size} height={size} viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className={className}><line x1="2" y1="2" x2="22" y2="22"></line><path d="M18.89 13.23A7.12 7.12 0 0 0 19 12v-2"></path><path d="M5 10v2a7 7 0 0 0 12 5l-2.7-2.7a4.5 4.5 0 0 1-6.3-6.3L5 6.13Z"></path><path d="M15 9.34V5a3 3 0 0 0-5.68-1.33l1.67 1.67H11a1 1 0 0 1 1 1v2.34Z"></path><line x1="12" y1="19" x2="12" y2="22"></line></svg>

        );


        function App() {

            const [gameState, setGameState] = useState('start'); // start, playing, game_complete

            const [isListening, setIsListening] = useState(false);

            const [transcript, setTranscript] = useState('');

            const [solvedWords, setSolvedWords] = useState([]);

            const [animatingWords, setAnimatingWords] = useState([]);

            const [errorMsg, setErrorMsg] = useState('');

            

            const recognitionRef = useRef(null);

            

            // 使用 useRef 來儲存狀態，讓語音辨識事件能讀取到最新值，不受閉包限制

            const solvedRef = useRef(solvedWords);

            const manualStopRef = useRef(true); 

            const gameStateRef = useRef(gameState);


            useEffect(() => { solvedRef.current = solvedWords; }, [solvedWords]);

            useEffect(() => { gameStateRef.current = gameState; }, [gameState]);


            // --- 判斷過關邏輯 ---

            useEffect(() => {

                if (solvedWords.length === WORDS.length && WORDS.length > 0) {

                    // 自動停止麥克風

                    manualStopRef.current = true;

                    if (recognitionRef.current) {

                        recognitionRef.current.stop();

                    }

                    setIsListening(false);


                    // 延遲 2.5 秒等待最後一個動畫跑完，再切換到過關畫面

                    const timer = setTimeout(() => {

                        setGameState('game_complete');

                    }, 2500);

                    return () => clearTimeout(timer);

                }

            }, [solvedWords]);


            // --- 初始化語音辨識 ---

            useEffect(() => {

                const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;

                

                if (!SpeechRecognition) {

                    setErrorMsg('您的瀏覽器不支援語音辨識功能，請使用 Google Chrome。');

                    return;

                }


                const recognition = new SpeechRecognition();

                recognition.continuous = true;

                recognition.interimResults = false;

                recognition.lang = 'en-US'; 


                recognition.onstart = () => {

                    setIsListening(true);

                    setErrorMsg('');

                };


                recognition.onresult = (event) => {

                    const current = event.resultIndex;

                    const resultText = event.results[current][0].transcript.toLowerCase().trim();

                    setTranscript(resultText);

                    checkMatch(resultText);

                };


                recognition.onerror = (event) => {

                    console.error('Speech recognition error', event.error);

                    if (event.error === 'not-allowed') {

                        setErrorMsg('請允許瀏覽器使用麥克風權限。');

                        manualStopRef.current = true; 

                    }

                    setIsListening(false);

                };


                // ★ 真・連續收音邏輯：如果不是使用者手動關閉，且遊戲還在進行中，就自動重啟

                recognition.onend = () => {

                    setIsListening(false);

                    if (!manualStopRef.current && gameStateRef.current === 'playing') {

                        try {

                            recognition.start();

                        } catch (e) {

                            console.error("重啟語音失敗", e);

                        }

                    }

                };


                recognitionRef.current = recognition;


                return () => {

                    manualStopRef.current = true;

                    if (recognitionRef.current) {

                        recognitionRef.current.stop();

                    }

                };

            }, []);


            const checkMatch = useCallback((spokenText) => {

                const spokenWords = spokenText.replace(/[^a-z0-9\s]/g, '').split(/\s+/);

                

                WORDS.forEach(word => {

                    const isMatched = spokenWords.includes(word.en) || 

                                      (word.aliases && word.aliases.some(alias => spokenWords.includes(alias)));

                    

                    if (isMatched && !solvedRef.current.includes(word.en)) {

                        triggerSolveAnimation(word.en);

                    }

                });

            }, []);


            const triggerSolveAnimation = (wordEn) => {

                setAnimatingWords(prev => [...prev, wordEn]);

                

                setTimeout(() => {

                    setSolvedWords(prev => [...prev, wordEn]);

                }, 600);


                setTimeout(() => {

                    setAnimatingWords(prev => prev.filter(w => w !== wordEn));

                }, 2000);

            };


            const toggleListening = () => {

                if (errorMsg && !recognitionRef.current) return;


                if (isListening) {

                    manualStopRef.current = true; 

                    recognitionRef.current.stop();

                } else {

                    manualStopRef.current = false; 

                    try {

                        recognitionRef.current.start();

                    } catch (e) {

                        console.error(e);

                    }

                }

            };


            const startGame = () => {

                setGameState('playing');

                setSolvedWords([]);

                setTranscript('');

                setAnimatingWords([]);

                

                // 啟動遊戲時自動開啟麥克風

                manualStopRef.current = false;

                setTimeout(() => {

                    if (recognitionRef.current && !isListening) {

                        try {

                            recognitionRef.current.start();

                        } catch (e) {}

                    }

                }, 500);

            };


            const resetGame = () => {

                setGameState('start');

                setSolvedWords([]);

                setTranscript('');

                setAnimatingWords([]);

                manualStopRef.current = true;

                if (recognitionRef.current) {

                    recognitionRef.current.stop();

                }

            };


            return (

                <div className="min-h-screen bg-slate-50 text-slate-800 font-sans prevent-select p-4 md:p-8">

                    

                    {/* Header */}

                    <header className="bg-white shadow-sm p-4 flex items-center justify-center rounded-2xl mb-8 max-w-4xl mx-auto">

                        <div className="flex items-center gap-2 font-bold text-xl text-blue-600">

                            <IconMic className="text-blue-500" />

                            英語語音達人 - 朗誦篇

                        </div>

                    </header>


                    <main className="max-w-4xl mx-auto flex flex-col items-center justify-center min-h-[60vh]">

                        

                        {/* --- 狀態 1：開始畫面 --- */}

                        {gameState === 'start' && (

                            <div className="text-center space-y-8 animate-in fade-in duration-500">

                                <div className="bg-blue-100 w-32 h-32 rounded-full flex items-center justify-center mx-auto mb-6 shadow-inner">

                                    <IconMic size={64} className="text-blue-500" />

                                </div>

                                <h1 className="text-4xl md:text-5xl font-extrabold text-slate-800 mb-4">

                                    準備好挑戰語音朗誦了嗎？

                                </h1>

                                <p className="text-lg text-slate-600 mb-8 max-w-md mx-auto">

                                    點擊開始後，請允許使用麥克風，並直接對著螢幕唸出卡片上的單字英文！

                                </p>

                                <button 

                                    onClick={startGame}

                                    className="bg-blue-600 hover:bg-blue-700 text-white text-xl font-bold py-4 px-12 rounded-full shadow-lg shadow-blue-200 transform transition hover:scale-105 active:scale-95 flex items-center gap-2 mx-auto"

                                >

                                    <IconPlay fill="currentColor" /> 開始挑戰

                                </button>

                            </div>

                        )}


                        {/* --- 狀態 2：遊戲進行中 --- */}

                        {gameState === 'playing' && (

                            <div className="w-full animate-in zoom-in duration-300">

                                {/* 麥克風控制區 */}

                                <div className="flex flex-col items-center justify-center mb-10">

                                    <button

                                        onClick={toggleListening}

                                        className={`w-20 h-20 flex items-center justify-center rounded-full text-white transition-all shadow-lg

                                            ${isListening ? 'bg-red-500 mic-active' : 'bg-blue-500 hover:bg-blue-600'}

                                            ${errorMsg ? 'opacity-50 cursor-not-allowed' : ''}

                                        `}

                                    >

                                        {isListening ? <IconMic size={36} /> : <IconMicOff size={36} />}

                                    </button>

                                    <p className="mt-4 font-bold text-lg text-slate-600">

                                        {isListening ? '正在持續聆聽您的發音...' : '點擊麥克風開始收音'}

                                    </p>

                                    

                                    {/* 即時語音辨識結果 */}

                                    <div className="h-8 mt-2 text-slate-400">

                                        {transcript && <span>系統聽到: <strong className="text-blue-500">{transcript}</strong></span>}

                                    </div>

                                    

                                    {errorMsg && (

                                        <p className="mt-2 text-red-500 font-semibold">{errorMsg}</p>

                                    )}

                                </div>


                                {/* 單字卡片網格 */}

                                <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-5 gap-4 md:gap-6">

                                    {WORDS.map((word) => {

                                        const isSolved = solvedWords.includes(word.en);

                                        const isAnimating = animatingWords.includes(word.en);

                                        

                                        return (

                                            <div 

                                                key={word.en}

                                                className={`relative h-40 flex flex-col items-center justify-center rounded-2xl border-4 transition-colors duration-300

                                                    ${isSolved ? 'bg-green-100 border-green-400 text-green-700 shadow-inner' : 'bg-white border-slate-200 text-slate-700 shadow-md'}

                                                `}

                                            >

                                                <div className={`flex flex-col items-center justify-center w-full h-full ${isAnimating ? 'animate-fly-up' : ''}`}>

                                                    {isSolved ? (

                                                        <span className="text-4xl md:text-5xl font-black">{word.en}</span>

                                                    ) : (

                                                        <>

                                                            <span className="text-5xl mb-2">{word.symbol}</span>

                                                            <span className="text-xl font-bold">{word.zh}</span>

                                                        </>

                                                    )}

                                                </div>

                                            </div>

                                        );

                                    })}

                                </div>

                            </div>

                        )}


                        {/* --- 狀態 3：過關畫面 --- */}

                        {gameState === 'game_complete' && (

                            <div className="text-center space-y-6 animate-in zoom-in duration-700">

                                <div className="bg-yellow-100 w-40 h-40 rounded-full flex items-center justify-center mx-auto mb-6 shadow-xl">
